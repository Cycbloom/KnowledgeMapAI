### 2.4 错误处理统一化

#### 问题分析
- 存在多种错误处理方式
- 39 个文件中有 101 处 console.log/error/warn 调用

#### 重构方案

```typescript
// 1. 创建统一错误处理服务
// services/errorHandling.ts
export class ErrorHandler {
  private addMessage: (msg: Message) => void;
  
  constructor(addMessage: (msg: Message) => void) {
    this.addMessage = addMessage;
  }
  
  handle(error: unknown, context?: string) {
    const message = this.parseError(error);
    console.error(`[${context}]`, error); // 开发环境
    this.addMessage({ type: 'error', content: message });
  }
  
  private parseError(error: unknown): string {
    if (error instanceof Error) return error.message;
    if (typeof error === 'string') return error;
    return '操作失败，请稍后重试';
  }
}

// 2. 创建错误处理 Hook
export const useErrorHandling = () => {
  const { addMessage } = useMessageStore();
  const handler = useMemo(() => new ErrorHandler(addMessage), []);
  
  const withErrorHandling = async <T>(
    fn: () => Promise<T>,
    context: string
  ): Promise<T | null> => {
    try {
      return await fn();
    } catch (error) {
      handler.handle(error, context);
      return null;
    }
  };
  
  return { handleError: handler.handle.bind(handler), withErrorHandling };
};
```

```typescript
// api/services/learningPathService.ts
export class LearningPathService {
  async generatePath(userId: string, graphId: string) {

      2. 新知识学习需基于已掌握的前置知识
```

```typescript
// 使用 Supabase Realtime 实现实时对战
// src/hooks/useBattle.ts
export const useBattle = (battleId: string) => {
  const [battle, setBattle] = useState<Battle | null>(null);
  
  useEffect(() => {
    const channel = supabase
      .channel(`battle:${battleId}`)
      .on('postgres_changes', {
        event: 'UPDATE',
        schema: 'public',
        table: 'battles',
        filter: `id=eq.${battleId}`
      }, (payload) => {
        setBattle(payload.new as Battle);
      })
      .subscribe();
      
    return () => { channel.unsubscribe(); };
  }, [battleId]);
  
  const submitAnswer = async (answer: string) => {
    await api.battles.submitAnswer(battleId, answer);
  };
  
  return { battle, submitAnswer };
};
```

### 3.5 🔄 知识图谱协作编辑（中优先级）

```typescript
// 使用 CRDT (Conflict-free Replicated Data Types) 实现协作
// 安装: npm install yjs y-websocket

// src/collaboration/CollaborationProvider.tsx
import * as Y from 'yjs';
import { WebsocketProvider } from 'y-websocket';

export const CollaborationProvider = ({ 
  graphId, 
  children 
}: { graphId: string; children: React.ReactNode }) => {
  const [ydoc] = useState(() => new Y.Doc());
  const [provider, setProvider] = useState<WebsocketProvider | null>(null);
  
  useEffect(() => {
    const wsProvider = new WebsocketProvider(
      'wss://your-collab-server.com',
      graphId,
      ydoc
    );
    setProvider(wsProvider);
    
    return () => { wsProvider.destroy(); };
  }, [graphId]);
  
  return (
    <CollaborationContext.Provider value={{ ydoc, provider }}>
      {children}
    </CollaborationContext.Provider>
  );
};

// 节点协作编辑
const useCollaborativeNodes = (graphId: string) => {
  const { ydoc } = useCollaboration();
  const nodesMap = ydoc.getMap(`nodes-${graphId}`);
  
  const addNode = (node: Node) => {
    nodesMap.set(node.id, node);
  };
  
  const updateNode = (id: string, updates: Partial<Node>) => {
    const existing = nodesMap.get(id);
    nodesMap.set(id, { ...existing, ...updates });
  };
  
  // 自动同步到所有协作者
  return { nodes: Array.from(nodesMap.values()), addNode, updateNode };
};
```
